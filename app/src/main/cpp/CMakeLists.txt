cmake_minimum_required(VERSION 3.22)
project(PhoenixMuseum LANGUAGES C CXX)
add_library(museum SHARED museum.cpp)
target_compile_features(museum PRIVATE cxx_std_20)
target_compile_options(museum PRIVATE -O2 -fno-fast-math)
target_link_options(museum PRIVATE -Wl,-z,max-page-size=16384)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
target_include_directories(museum PRIVATE "${REPO_ROOT}/external/phoenix/native_runtime" "${REPO_ROOT}/external/phoenix/third_party/nlohmann")
target_link_libraries(museum GLESv3 android log dl)
