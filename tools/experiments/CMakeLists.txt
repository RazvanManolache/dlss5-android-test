cmake_minimum_required(VERSION 3.22)
project(PhoenixMobileExperiments LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PHOENIX "${APP_ROOT}/external/phoenix")
add_executable(capture_projection capture_projection.cpp)
target_include_directories(capture_projection PRIVATE "${PHOENIX}/native_runtime" "${PHOENIX}/third_party/nlohmann")
target_compile_options(capture_projection PRIVATE -O2 -fno-fast-math)
target_link_libraries(capture_projection PRIVATE dl)
add_executable(sequence_probe sequence_probe.cpp)
target_include_directories(sequence_probe PRIVATE "${PHOENIX}/native_runtime" "${PHOENIX}/third_party/nlohmann")
target_compile_options(sequence_probe PRIVATE -O2 -fno-fast-math)
target_link_libraries(sequence_probe PRIVATE dl)
if(QNN_SDK)
  foreach(probe qnn_projection_probe qnn_ffn_probe)
    add_executable(${probe} ${probe}.cpp)
    target_include_directories(${probe} PRIVATE "${QNN_SDK}/include/QNN" "${PHOENIX}/third_party/nlohmann")
    target_compile_options(${probe} PRIVATE -O2 -fno-fast-math)
    target_link_libraries(${probe} PRIVATE dl)
  endforeach()
endif()
